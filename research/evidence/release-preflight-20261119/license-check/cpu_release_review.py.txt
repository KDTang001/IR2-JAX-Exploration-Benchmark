def run(name: str, argv: list[str], timeout: int = 300, extra_env: dict[str, str] | None = None) -> str:
    """Retain the exact command, bounded CPU status and output before failing."""
    started = time.time()
    (root / "current.json").write_text(json.dumps({"phase": name, "started_epoch": started}))
    current = env if extra_env is None else env | extra_env
    result = subprocess.run(argv, cwd=work, env=current, stdin=subprocess.DEVNULL,
                            capture_output=True, text=True, timeout=timeout, check=False)
    output = result.stdout + result.stderr
    (root / (name + ".log")).write_text(output, encoding="utf-8")
    commands.append({"phase": name, "argv": argv, "exit_code": result.returncode, "seconds": time.time() - started})
    (root / "commands.json").write_text(json.dumps(commands, indent=2) + "\n")
    print(json.dumps(commands[-1]), flush=True)
    if result.returncode:
        (root / "cpu-result.json").write_text(json.dumps({"complete": False, "failed_phase": name, "exit_code": result.returncode}))
        raise SystemExit(result.returncode)
    return output
