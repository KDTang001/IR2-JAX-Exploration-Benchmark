class WorkerInstance(TypedDict):
    pid: int
    start_ticks: int
    argv: list[str]


def stamp() -> str:
    return datetime.now(UTC).isoformat()


def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def source_digest() -> str:
    value = hashlib.sha256()
    for path in sorted((SOURCE / 'src').rglob('*.py')):
        value.update(str(path.relative_to(SOURCE)).encode())
        value.update(path.read_bytes())
    return value.hexdigest()


def save(path: Path, data: Json) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_name(path.name + '.tmp')
    temp.write_text(json.dumps(data, indent=2) + '\n')
    temp.replace(path)


def stage(name: str) -> None:
    save(STATE, {'status': 'working', 'stage': name, 'updated_utc': stamp()})
    print(json.dumps({'stage': name, 'utc': stamp()}), flush=True)


def worker_instance(pid: int) -> WorkerInstance | None:
    """Read active process startticks and argv without authorizing reused or unrelated PIDs."""
    try:
        fields = (PROC / str(pid) / 'stat').read_text().rsplit(')', 1)[1].split()
        argv = [part.decode() for part in (PROC / str(pid) / 'cmdline').read_bytes().split(b'\0') if part]
    except FileNotFoundError:
        return None
    authorized = (fields[0] != 'Z' and any(Path(arg).name == 'phase5_full_episodes.py' for arg in argv)
                  and '--output' in argv and argv.index('--output') + 1 < len(argv)
                  and argv[argv.index('--output') + 1] == str(ROOT / 'long'))
    return {'pid': pid, 'start_ticks': int(fields[19]), 'argv': argv} if authorized else None


def same_instance(worker: WorkerInstance) -> bool:
    return worker_instance(worker['pid']) == worker


def signal_same_worker(worker: WorkerInstance, sig: signal.Signals) -> bool:
    """Use a kernel PID handle so PID reuse between identity check and signal is harmless."""
    try:
        descriptor = os.pidfd_open(worker['pid'])
    except ProcessLookupError:
        return False
    with ExitStack() as stack:
        stack.callback(os.close, descriptor)
        if not same_instance(worker):
            return False
        try:
            signal.pidfd_send_signal(descriptor, sig)
        except ProcessLookupError:
            return False
        return True


def enforce_deadline(workers: list[WorkerInstance]) -> None:
    """Only at the actual deadline, stop exactly the captured live worker instances."""
    assert time.time() >= DEADLINE
    actions = []
    for worker in workers:
        if signal_same_worker(worker, signal.SIGTERM):
            actions.append({'pid': worker['pid'], 'signal': 'SIGTERM', 'start_ticks': worker['start_ticks']})
    finish = time.monotonic() + 10
    while any(same_instance(worker) for worker in workers) and time.monotonic() < finish:
        time.sleep(0.2)
    for worker in workers:
        if signal_same_worker(worker, signal.SIGKILL):
            actions.append({'pid': worker['pid'], 'signal': 'SIGKILL', 'start_ticks': worker['start_ticks']})
    save(ROOT / 'control/deadline-enforcement.json', {'enforced_utc': stamp(), 'captured_workers': workers,
         'actions': actions, 'record_policy': 'partial results/traces/logs retained without modification'})


def read_when_ready(path: Path) -> Json:
    """Retry incomplete producer JSON writes; never turn malformed data into a gate."""
    while time.time() < DEADLINE:
        if path.exists():
            try:
                return json.loads(path.read_text())
            except (json.JSONDecodeError, UnicodeError):
                pass
        time.sleep(15)
    raise TimeoutError(f'Overall deadline reached waiting for {path}')


def task_rows(path: Path) -> list[Json]:
    return [json.loads(line) for line in path.read_text().splitlines() if line.strip()]


def record_hashes() -> Json:
    return {path.name: digest(path) for path in sorted((ROOT / 'long/results').glob('*.json'))}


def run_job(argv: list[str], log: Path, timeout: int) -> None:
    """Wait for subprocess exit and terminate its entire group on an actual deadline."""
    environment = os.environ.copy()
    environment.update(PYTHONPATH=str(SOURCE / 'src'), JAX_PLATFORMS='cpu',
                       OMP_NUM_THREADS='1', OPENBLAS_NUM_THREADS='1', MKL_NUM_THREADS='1', MPLBACKEND='Agg')
    remaining = min(timeout, DEADLINE-time.time())
    assert remaining > 0
    with log.open('w') as stream:
        process = subprocess.Popen(argv, cwd=SOURCE, env=environment, stdout=stream,
                                   stderr=subprocess.STDOUT, start_new_session=True)
        save(ROOT / 'control/post-long-active-child.json', {'pid': process.pid, 'argv': argv, 'log': str(log), 'started_utc': stamp()})
        try:
            returncode = process.wait(timeout=remaining)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid, signal.SIGTERM)
            try:
                process.wait(timeout=10)
            except subprocess.TimeoutExpired:
                os.killpg(process.pid, signal.SIGKILL)
                process.wait()
            raise
        save(ROOT / 'control/post-long-active-child.json', {'pid': process.pid, 'returncode': returncode, 'finished_utc': stamp(), 'log': str(log)})
        assert returncode == 0, f'{argv[3:]} failed exit{returncode}; see {log}'

def validate_audit(path: Path, identities: set[str]) -> Json:
    payload = json.loads(path.read_text())
    assert payload['status'] == 'passed' and payload['tasks'] == len(identities)
    assert payload['execution_error_records'] == 0
    assert len(payload['results']) == len(identities)
    assert {row['result_id'] for row in payload['results']} == identities
    return payload


def quiescence() -> Json:
    """Require no benchmark/oracle workers and two idle GPU snapshots before publication."""
    snapshots = []
    for _ in range(2):
        processes = subprocess.run(['ps', '-eo', 'pid,args'], capture_output=True, text=True, check=True).stdout
        forbidden = ('scripts/phase5_full_episodes.py', 'scripts/audit_baseline_evaluation.py',
                     'phase5_performance.py', 'performance_v2.py worker', 'arithmetic_provenance.py', 'aggregate_long.py')
        busy = [line for line in processes.splitlines() if any(token in line for token in forbidden)]
        assert not busy, f'Measurement/oracle subprocesses still running: {busy}'
        gpu = subprocess.run(['nvidia-smi', '--query-gpu=name,uuid,utilization.gpu,memory.used,memory.total',
                              '--format=csv,noheader,nounits'], capture_output=True, text=True, check=True).stdout.strip()
        apps = subprocess.run(['nvidia-smi', '--query-compute-apps=pid,process_name,used_memory',
                               '--format=csv,noheader,nounits'], capture_output=True, text=True, check=True).stdout.strip()
        assert not apps and int(gpu.split(',')[-3]) <= 5, f'GPU is not quiet: {gpu}; {apps}'
        snapshots.append({'utc': stamp(), 'gpu': gpu, 'compute_processes': apps, 'processes': processes})
        time.sleep(1)
    return snapshots

def main() -> None:
    if len(sys.argv) == 2 and sys.argv[1] == 'public':
        public_resume_report()
        return
    with (ROOT / 'control/post-long-supervisor.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        workers = [value for pid in WORKERS if (value := worker_instance(pid)) is not None]
        save(ROOT / 'control/known-long-worker-instances.json', {'captured_utc': stamp(), 'workers': workers})
        try:
            supervise(workers)
        except Exception as error:  # noqa: BROAD_EXCEPT_OK - CLI boundary preserves blocked evidence and exits nonzero.
            if time.time() >= DEADLINE:
                enforce_deadline(workers)
            save(STATE, {'status': 'blocked', 'updated_utc': stamp(), 'error_type': type(error).__name__,
                         'error': str(error), 'traceback': traceback.format_exc(), 'final_gate_created': GATE.exists()})
            traceback.print_exc()
            raise


if __name__ == '__main__':
    main()
