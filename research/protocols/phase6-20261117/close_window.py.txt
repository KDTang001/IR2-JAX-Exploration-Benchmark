"""Close the finished Phase 6 measurement window without touching earlier evidence."""

import hashlib
import json
import os
import pathlib
import subprocess
from datetime import datetime, timezone

r = pathlib.Path('/home/user/Tz/acceptance/phase6-prep-20261117')
window = json.loads((r / 'gpu-window.json').read_text())
summaries = {name: json.loads((r / f'{name}-summary.json').read_text())
             for name in ('profile', 'comparison', 'mini')}
assert [len(summaries[name]['cases']) for name in summaries] == [6, 6, 2]
assert all(row['exit_code'] == 0 for summary in summaries.values() for row in summary['cases'])
end = json.loads((r / 'mini-finished.json').read_text())['utc']
elapsed = datetime.fromisoformat(end).timestamp() - window['started_epoch']
assert elapsed < window['authorized_seconds']
workers = []
names = ('phase6_profile.py', 'phase6_compare.py', 'phase6_profile_supervisor.py', 'after_comparison.py')
processes = subprocess.run(['ps', '-eo', 'pid,args'], capture_output=True, text=True, check=True)
for line in processes.stdout.splitlines()[1:]:
    pid, command = line.strip().split(None, 1)
    if int(pid) not in (os.getpid(), os.getppid()) and str(r) in command and any(name in command for name in names):
        workers.append({'pid': int(pid), 'command': command})
assert not workers
gpu = subprocess.run(['nvidia-smi', '--query-compute-apps=pid,process_name,used_memory',
                      '--format=csv,noheader,nounits'], capture_output=True, text=True, check=True)
result = {
    'status': 'closed_complete', 'closed_utc': datetime.now(timezone.utc).isoformat(),
    'completed_utc': end, 'authorized_seconds': window['authorized_seconds'],
    'first_gpu_launch_to_completion_seconds': elapsed,
    'isolated_worker_wall_seconds_sum': sum(row['wall_seconds'] for summary in summaries.values() for row in summary['cases']),
    'completed_workers': {'profiles': 6, 'optimization_pairs': 6, 'full_episode_cost': 2},
    'active_phase6_workers': workers, 'gpu_processes_at_close': gpu.stdout.strip(),
    'unused_budget_reused': False, 'full240_pilot_started': False, 'training_started': False,
    'summary_hashes': {name: hashlib.sha256((r / f'{name}-summary.json').read_bytes()).hexdigest() for name in summaries},
    'worker_sources_at_close': {name: hashlib.sha256((r / name).read_bytes()).hexdigest()
                               for name in ('phase6_profile.py', 'phase6_compare.py', 'phase6_profile_supervisor.py', 'after_comparison.py')},
}
with (r / 'gpu-window-closed.json').open('x') as stream:
    stream.write(json.dumps(result, indent=2) + '\n')
print(json.dumps(result, indent=2))
